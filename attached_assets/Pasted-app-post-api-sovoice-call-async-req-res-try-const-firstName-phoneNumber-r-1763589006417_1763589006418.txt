app.post("/api/sovoice/call", async (req, res) => {
  try {
    const { firstName, phoneNumber } = req.body;
    
    if (!firstName || !phoneNumber) {
      return res.status(400).json({ 
        error: "firstName and phoneNumber are required" 
      });
    }

    const apiKey = process.env.PICATIC_API_KEYSOVOICENEW;
    
    if (!apiKey) {
      console.error("PICATIC_API_KEYSOVOICENEW is not set");
      return res.status(500).json({ 
        error: "API key not configured" 
      });
    }

    console.log(`Initiating SoVoice call for ${firstName} at ${phoneNumber}`);
    console.log(`Using API Key: ${apiKey.substring(0, 15)}...`);

    // Call SoVoice API
    try {
      const response = await fetch("https://sovoice.ai/api/sovoice/call", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${apiKey}`
        },
        body: JSON.stringify({
          firstName,
          phoneNumber,
          direction: 'outbound'
        })
      });

      console.log(`SoVoice API Response Status: ${response.status}`);
      
      const responseText = await response.text();
      console.log(`SoVoice API Response Body: ${responseText}`);
      
      let data;
      try {
        data = JSON.parse(responseText);
      } catch (e) {
        console.error("Failed to parse response as JSON:", responseText);
        data = { rawResponse: responseText };
      }

      if (!response.ok) {
        console.error("SoVoice API error:", data);
        return res.status(response.status).json({ 
          error: "Failed to initiate call",
          details: data
        });
      }

      console.log("SoVoice call initiated successfully:", data);

      return res.json({
        success: true,
        data
      });
    } catch (fetchError) {
      console.error("Fetch error:", fetchError);
      throw fetchError;
    }

  } catch (error) {
    console.error("Error calling SoVoice API:", error);
    return res.status(500).json({ 
      error: "Internal server error",
      message: error instanceof Error ? error.message : "Unknown error"
    });
  }
});
